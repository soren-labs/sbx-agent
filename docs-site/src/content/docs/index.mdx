import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid>
  <Card title="Session-first" icon="open-book">
    A Session is the durable conversation and work. Compute is a replaceable
    ExecutorLease; the Worktree and its checkpoints outlive any one sandbox.
  </Card>
  <Card title="Official CLIs" icon="rocket">
    SBX supervises the provider's own CLI (OpenCode is supported, Codex is
    experimental). It runs no model loop of its own.
  </Card>
  <Card title="Immutable ChangeSets" icon="code-branch">
    Captured changes are sealed with a subject digest. Delivery pushes exactly
    that subject, and merge is gated on the same digest.
  </Card>
  <Card title="One API" icon="setting">
    A single `/api` surface serves the Console, the Python SDK and the CLI, with
    one event journal and one watermark.
  </Card>
</CardGrid>
